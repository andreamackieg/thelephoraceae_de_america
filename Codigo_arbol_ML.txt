#Código para generar árboles mediante Máxima Verosimilitud. Desarrollado por Susana Magallón y editado por Andrea Mackie Greene.

#Debe instalarse RaxMl versión 8 de Alexandros Stamatakis y FigTree v1.4.4

#Activar el ambiente donde está MAFFT
micromamba activate base

#Alinear las secuencias con MAFFT (para menor cant. de secuencias):
mafft --auto --leavegappyregion --thread -1 --op 3.0 --ep 0.123 \
/home/andrea/Documentos/secuencias_curadas/Thelephoraceae_OTUs.fasta \
> /home/andrea/Documentos/secuencias_alineadas/Thelephoraceae_OTUs_alineado_mafft.fasta

#Alinear con muscle5 (para mayor cant. de secuencias):
muscle -threads 20 -super5 \
/home/andrea/Documentos/secuencias_curadas/Thelephoraceae_OTUs.fasta \
-output /home/andrea/Documentos/secuencias_alineadas/Thelephoraceae_OTUs_alineado_muscle5.fasta

#Activar ambiente biophyton:
micromamba activate convert_env

#Las secuencias FASTA ya alineadas deben estar en formato PHYLIP, para convertirlas:
python - <<EOF
from Bio import AlignIO
AlignIO.convert(
"/home/andrea/Documentos/secuencias_alineadas/Canada_alineado.fasta",
"fasta",
"/home/andrea/Documentos/secuencias_alineadas/Canada.phy",
"phylip-relaxed"
)
EOF


#Luego activar el ambiente de análisis (donde está figtree):
micromamba deactivate
micromamba activate figtree_env


#Para ejecutar RaxMl debe seleccionarse el modelo adecuado para los datos (puede ser en CIPRES), en este caso se usó GTR:
raxmlHPC-PTHREADS \
-f a \
-m GTRGAMMA \
-s /home/andrea/Documentos/secuencias_alineadas/Ecuador.phy \
-n arbol.prueba \
-w /home/andrea/Documentos/output \
-T 4 \
-p 12345 \
-x 12345 \
-# 100

#Luego abrir en FigTree:
figtree /home/andrea/Documentos/output/RAxML_bipartitions.arbol.prueba

#Comando para mapear bootstrap al aŕbol de ML:
raxmlHPC-PTHREADS \
-f b \
-m GTRGAMMA \
-s /home/andrea/Documentos/secuencias/Ecuador.phy \
-t /home/andrea/Documentos/output/RAxML_bestTree.arbol.prueba \
-z /home/andrea/Documentos/output/RAxML_bootstrap.arbol.prueba \
-n BS_TREE.arbol.prueba \
-w /home/andrea/Documentos/output \
-T 4

#Luego abrir en Figtree:
figtree /home/andrea/Documentos/output/RAxML_bipartitions.BS_TREE.arbol.prueba
